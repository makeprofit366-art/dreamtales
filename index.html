export default async function handler(req, res) { try { const apiKey = process.env.GEMINI_API_KEY; const { name, age, animal, len, wish, lang } = req.body; const prompt = ` Create a bedtime story for ${name}, age ${age}. Character: ${animal} Language: ${lang} Length: ${len} Extra: ${wish} `; const response = await fetch( "https://generativelanguage.googleapis.com/v1/models/gemini-pro:generateContent?key=" + apiKey, { method: "POST", headers: { "Content-Type": "application/json", }, body: JSON.stringify({ contents: [{ parts: [{ text: prompt }] }], }), } ); const data = await response.json(); res.status(200).json(data); } catch (error) { res.status(500).json({ error: "Error" }); } }
